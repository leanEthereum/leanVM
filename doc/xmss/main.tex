\begin{abstract}
As proposed in~\cite{DKKW25}, XMSS is a promising post-quantum replacement for BLS \cite{BLS}, the signature scheme used by Ethereum validators. The lack of native aggregation can be mitigated with a post-quantum snark. This technical note specifies an XMSS instance with the following properties:
\begin{itemize}
  \item \textbf{NIST security level~1}~\cite{NISTPQC}, $\approx 128$-bit (resp. $\approx 64$-bit) of classical (resp. quantum) security, in the Random Oracle model, ROM (resp. Quantum Random Oracle model, QROM).\footnote{The classical bound is proved (Section~\ref{sec:security}); the quantum one is a target, not a proved statement (Section~\ref{sec:quantum}).}
  \item \textbf{public key: 32 bytes}.
  \item \textbf{signature: 1208 bytes} (smaller than the 1280-byte minimum IPv6 MTU~\cite{RFC8200}).
  \item snark-friendly encoding, optimized for aggregation via leanVM~\cite{leanVM}, requiring 15 bits of grinding on average.
  \item 133 hashes per verification.
\end{itemize}
\end{abstract}

The key difference with \cite{DKKW25} is the use of short hash digests, of 16 bytes, following SLH-DSA~\cite{FIPS205} at level 1. The consequence is that the security proof takes place in the ROM, while \cite{DKKW25} is secure in the Standard Model (as a cost of bigger hash digests, close to 32 bytes).

\section{Definitions and notation}

\begin{definition}[Synchronized signature scheme]
A synchronized signature scheme with lifetime $L$ is a tuple $\SIG=(\Gen,\Sig,\Ver)$, where $\Gen$ and $\Sig$ are randomized and $\Ver$ is deterministic:
\[
  \Gen\longrightarrow(\pk,\sk),\qquad
  \Sig(\sk,\ep,m)\longrightarrow\sigma\in\Sigma\cup\{\bot\},\qquad
  \Ver(\pk,\ep,m,\sigma)\longrightarrow\{0,1\},
\]
where $\Sigma$ is the signature space, $m\in\bits{\lmsg}$, and $\ep\in\{0,\ldots,L-1\}$. Whenever $(\pk,\sk)$ is output by $\Gen$ and $\Sig(\sk,\ep,m)$ returns $\sigma\neq\bot$, correctness requires $\Ver(\pk,\ep,m,\sigma)=1$. For each key pair, $\Sig(\sk,\ep,\cdot)$ may be invoked at most once for any fixed epoch $\ep$.
\end{definition}

XMSS is a synchronized signature scheme in which epoch $\ep$ selects a Winternitz one-time key authenticated by leaf $\ep$ of a Merkle tree~\cite{RFC8391}. This specification sets $\lmsg=256$ and $L=2^h=2^{32}$.

Byte strings are concatenated with $\concat$. Bits and integer encodings are little endian. $\LE_r(a)$ is the unsigned $r$-bit encoding of $a$. All indices are zero based.

Let $\hash:\bits{*}\to\bits{256}$ be a cryptographic hash function, and let $\Truncate_{\nu\text{ bits}}$ keep the first $\nu$ output bits. The tweakable hash $\Th:\mathcal P\times\mathcal T\times\mathcal M\to\mathcal H$, with $\mathcal P=\bits{\lpar}$, $\mathcal T=\bits{\ltwk}$ and $\mathcal H=\bits{n}$, is
\[
  \Th(P,\tw,M)=\Truncate_{n\text{ bits}}\!\left(\hash(\tw\concat P\concat M)\right).
\]

\begin{definition}[Tweak encoding]
Let $\mathsf{protocol\_domain\_sep}=0$. For a one-byte type $t$ and unsigned 32-bit integers $p$ and $j$, define the 16-byte tweak
\[
\begin{aligned}
  \mathsf{enc}(t,p,j)={}&\LE_8(\mathsf{protocol\_domain\_sep})\concat\LE_8(t)\concat0^{16}\\
  &{}\concat\LE_{32}(p)\concat0^{32}\concat\LE_{32}(j).
\end{aligned}
\]
\end{definition}

Define
\[
\begin{aligned}
  \mathsf{tweak}_{\mathrm{chain}}(\ep,i,k) &= \mathsf{enc}(1,2^wi+k-1,\ep), &&0\leq i<v,\ 1\leq k\leq2^w-1,\\
  \mathsf{tweak}_{\mathrm{leaf}}(\ep) &= \mathsf{enc}(2,0,\ep),\\
  \mathsf{tweak}_{\mathrm{merkle}}(\ell,j) &= \mathsf{enc}(3,\ell,j), &&1\leq\ell\leq h,\ 0\leq j<2^{h-\ell},\\
  \mathsf{tweak}_{\mathrm{enc}}(\ep) &= \mathsf{enc}(4,0,\ep).
\end{aligned}
\]

The target-sum code is~\cite{DKKW25}
\[
  \mathcal C=\left\{(x_0,\ldots,x_{v-1})\in\{0,\ldots,2^w - 1\}^{v}:\sum_{i=0}^{v-1}x_i=T\right\}.
\]
Define $\IncEnc(P,m,\rho,\ep)$ by computing
\[
  D=\Th\!\left(P,\mathsf{tweak}_{\mathrm{enc}}(\ep),m\concat\rho\concat 0^{64}\right).
\]
Write $D=D_0\concat D_1$, where $D_0$ and $D_1$ are $n/2$-bit strings. For $q\in\{0,1\}$, let $d_q\in\{0,\ldots,2^{n/2}-1\}$ be the integer whose $n/2$-bit little-endian encoding is $D_q$. For $0\leq r<v/2$, set
\[
  x_{qv/2+r}=\left\lfloor\frac{d_q}{2^{wr}}\right\rfloor\bmod 2^w.
\]
Return $x=(x_0,\ldots,x_{v-1})$ if bit $wv/2=63$ of both $d_0$ and $d_1$ is zero and $x\in\mathcal C$; otherwise return $\bot$.

\begin{definition}[Hash chain]
For a chain index $i$, epoch $\ep$, start index $k$, number of steps $s$, and value $x\in\mathcal H$, define $\Chain_{\Th,i,\ep}(P,k,s,x)$ recursively by
\[
  \Chain_{\Th,i,\ep}(P,k,0,x)=x
\]
and, for $s\geq1$,
\[
  \Chain_{\Th,i,\ep}(P,k,s,x)=\Th\!\left(P,\mathsf{tweak}_{\mathrm{chain}}(\ep,i,k+s),\Chain_{\Th,i,\ep}(P,k,s-1,x)\right).
\]
The valid range is $0\leq k\leq2^w-1$ and $0\leq s\leq2^w-1-k$.
\end{definition}

The one-time public key and its Merkle leaf are
\[
\begin{aligned}
  \mathit{pk}_{\ep,i} &= \Chain_{\Th,i,\ep}(P,0,2^w-1,\mathit{sk}_{\ep,i}), &&0\leq i<v,\\
  X_{0,\ep} &= \Th\!\left(P,\mathsf{tweak}_{\mathrm{leaf}}(\ep),\mathit{pk}_{\ep,0}\concat\cdots\concat\mathit{pk}_{\ep,v-1}\right).
\end{aligned}
\]

\section{Key generation}

The algorithm $\Gen$ samples a master secret $S\getsr\bits{256}$ and derives
\[
  P=\Th\!\left(0^{\lpar},\mathsf{enc}(10,0,0),S\right),
  \qquad
  \mathit{sk}_{\ep,i}=\Th\!\left(P,\mathsf{enc}(0,i,\ep),S\right),
  \qquad 0\leq\ep<L,\quad 0\leq i<v.
\]
It computes and stores every chain value
\[
  C_{\ep,i,k}=\Chain_{\Th,i,\ep}(P,0,k,\mathit{sk}_{\ep,i}),
  \qquad 0\leq k<2^w.
\]
Thus $C_{\ep,i,0}=\mathit{sk}_{\ep,i}$ and $C_{\ep,i,2^w-1}=\mathit{pk}_{\ep,i}$. For $1\leq\ell\leq h$, define
\[
  X_{\ell,j}=\Th\!\left(P,\mathsf{tweak}_{\mathrm{merkle}}(\ell,j),X_{\ell-1,2j}\concat X_{\ell-1,2j+1}\right),
  \qquad 0\leq j<2^{h-\ell}.
\]
Set $\rootnode=X_{h,0}$ and return
\[
  \pk=(\rootnode,P),
  \qquad
  \sk=\left(S,P,(C_{\ep,i,k})_{\substack{0\leq\ep<L,\,0\leq i<v,\\0\leq k<2^w}},(X_{\ell,j})_{\substack{0\leq\ell\leq h,\\0\leq j<2^{h-\ell}}}\right).
\]

The public key is serialized as $\rootnode\concat P$.

\begin{remark}[Precomputed secret key]
The secret key caches the full Merkle tree and every intermediate chain value. Signing reads these tables without hashing them again. Implementations may recompute cached values from $S$ and $P$; those computations contribute to their hash-query cost.
\end{remark}

\begin{remark}[Restricted key generation]
Key generation may be restricted to a smaller interval of epochs by replacing subtrees outside that interval with filler nodes $\Th(P,\mathsf{enc}(11,\ell,j),S)$ at height $\ell$ and index $j$. The resulting key signs only epochs in that interval. This variant is outside the theorem of Section~\ref{sec:security}.
\end{remark}

\section{Signing}

$\Sig(\sk,\ep,m)$ returns $\bot$ if epoch $\ep$ has already been used and otherwise marks it used. It makes at most $\amax=2^{23}$ attempts. In each attempt it independently samples $\rho\getsr\bits{\lrnd}$ and computes $x=\IncEnc(P,m,\rho,\ep)$. If $x\neq\bot$, it immediately computes
\[
  \sigma_{\mathrm{OTS},i}=C_{\ep,i,x_i},
  \qquad 0\leq i<v,
\]
and the authentication path
\[
  A_\ell=X_{\ell,\,\left\lfloor\ep/2^\ell\right\rfloor\mathbin{\mathsf{xor}}1},
  \qquad 0\leq\ell<h.
\]
and returns
\[
  \sigma=(\rho,\sigma_{\mathrm{OTS}},\mathit{path}_{\ep}),
  \qquad
  \mathit{path}_{\ep}=(A_0,\ldots,A_{h-1}),
\]
serialized as $\sigma_{\mathrm{OTS},0}\concat\cdots\concat\sigma_{\mathrm{OTS},v-1}\concat\rho\concat A_0\concat\cdots\concat A_{h-1}$. If all $\amax$ attempts produce $x=\bot$, it returns $\bot$.

Consequently, signing queries the random oracle only when evaluating $\IncEnc$, once per attempt. Reading the stored chain values and authentication path makes no random-oracle query.

\section{Verification}

$\Ver(\pk,\ep,m,\bot)=0$. For $\sigma\neq\bot$, $\Ver(\pk,\ep,m,\sigma)$ parses $\pk=(\rootnode,P)$ and $\sigma=(\rho,(y_i)_i,(A_\ell)_\ell)$, then:
\begin{enumerate}[leftmargin=2em]
\item Compute $x=\IncEnc(P,m,\rho,\ep)$ and reject if $x=\bot$.
\item For every $0\leq i<v$, compute $\mathit{pk}'_{\ep,i}=\Chain_{\Th,i,\ep}(P,x_i,2^w-1-x_i,y_i)$.
\item Set $Z_0=\Th(P,\mathsf{tweak}_{\mathrm{leaf}}(\ep),\mathit{pk}'_{\ep,0}\concat\cdots\concat\mathit{pk}'_{\ep,v-1})$.
\item For $0\leq\ell<h$, let $j=\lfloor\ep/2^{\ell+1}\rfloor$. If bit $\ell$ of $\ep$ is zero, set $Z_{\ell+1}=\Th(P,\mathsf{tweak}_{\mathrm{merkle}}(\ell+1,j),Z_\ell\concat A_\ell)$; otherwise set $Z_{\ell+1}=\Th(P,\mathsf{tweak}_{\mathrm{merkle}}(\ell+1,j),A_\ell\concat Z_\ell)$.
\item Accept if and only if $Z_h=\rootnode$.
\end{enumerate}

\section{Security}
\label{sec:security}

\subsection{Classical security}

\begin{definition}[strong unforgeability in the ROM]
Let $\SIG=(\Gen,\Sig,\Ver)$ be a synchronized signature scheme whose algorithms use a hash function $\hash:\bits{*}\to\bits{256}$. Consider the following game between a signer and an adversary $\mathcal A$ (an arbitrary probabilistic algorithm with unbounded running time and memory), with $\hash$ sampled as a random oracle, that both signer and adversary can query. The signer first runs $(\pk,\sk)\gets\Gen$ and gives $\pk$ to $\mathcal A$. The adversary may then adaptively take any of the following actions:
\begin{enumerate}[leftmargin=2em]
  \item Query the random oracle on any input and receive its 256-bit output.
  \item Submit a message $m\in\bits{256}$ and an epoch $\ep\in\{0,\ldots,L-1\}$ that it has not used in an earlier signing query, and receive $\sigma\gets\Sig(\sk,\ep,m)$ from the signer.
  \item Terminate with a claimed forgery $(m^*,\ep^*,\sigma^*)$.
\end{enumerate}
The adversary wins if $\Ver(\pk,\ep^*,m^*,\sigma^*)=1$ and the signer did not return $\sigma^*$ in response to a signing query for $(m^*,\ep^*)$, meaning:
\begin{itemize}[leftmargin=2em]
  \item if the adversary never queried a signature for $(m^*,\ep^*)$;
  \item or it did, but the signer's answer was not $\sigma^*$.
\end{itemize}

Call $\mathcal A$ $q$-bounded if the experiment makes at most $q$ random-oracle queries on every execution, counting those of key generation, signing, and the final verification of the claimed forgery. We say that $\SIG$ has $x$ bits of classical strong unforgeability in the ROM if every $q\geq1$ and every $q$-bounded $\mathcal A$ satisfy
\[
  \Pr[\mathcal A\text{ wins}]\leq\frac{q}{2^{x}}.
\]
\end{definition}

\begin{remark}[Why not 128 bits?]
The following classical attack demonstrates that security $\leq 127,3$ bits. The same idea also applies to SLH-DSA.
\begin{enumerate}[leftmargin=2em]
  \item \textbf{Get a signature.} Ask for a signature and read its encoding $x$.
  \item \textbf{Find a nearby encoding.} For a new message at the same epoch, try $2^{121}$ distinct randomness values. Seek an encoding that decreases one digit $x_i$ by one and increases another digit $x_j$ by one, leaving the rest unchanged. The sum stays 195. At least 28 digits are nonzero, and at least 15 are below 7, so there are at least $28\cdot14=392$ choices of $(i,j)$ with $i\neq j$. Each choice specifies one 128-bit digest. A trial therefore succeeds with probability at least $392/2^{128}$, and independent trials give
  \[
    \begin{aligned}
      \Pr[\text{encoding found}] &\geq 1-\left(1-\frac{392}{2^{128}}\right)^{2^{121}} \geq 1-\exp\!\left(-\frac{392\cdot2^{121}}{2^{128}}\right)
       \approx0.953.
    \end{aligned}
  \]
  \item \textbf{Go back one chain step.} For the selected chain $i$, try $2^{124}$ distinct preimages of its signature value under the preceding chain step. A random guess has two ways to succeed: guess the genuine predecessor, or find another input with the same hash, each contributing about $2^{-128}$. The guesses contain the genuine predecessor with probability $2^{124}/2^{128}=1/16$. Otherwise, they find a collision with probability about $1-e^{-1/16}$. Thus
  \[
    \Pr[\text{preimage found}]\approx\frac{1}{16}+\frac{15}{16}\left(1-e^{-1/16}\right)\approx0.1193.
  \]
  \item \textbf{Forge.} Use the preimage on chain $i$ and hash forward once on chain $j$. Reuse the other signature values and the authentication path.
\end{enumerate}
The total success probability is at least about $0.953\cdot0.1193\approx0.114$, so
\[
  \frac{q}{\Pr[\text{success}]}\lesssim\frac{2^{121}+2^{124}}{0.953\cdot0.1193}\approx1.24\cdot2^{127}\approx2^{127.3}.
\]
\end{remark}

\subsection{Quantum security}
\label{sec:quantum}

Only the classical bound of Section~\ref{sec:security} is proved. We expect $\approx64$ quantum bits, NIST category~1, which FIPS~205 assigns to the SLH-DSA parameter sets with a 128-bit digest~\cite{FIPS205}. The XMSS layers of SPHINCS+ have a QROM bound against an adversary that does not choose the signed messages~\cite{HK22}, and hash-then-sign reduces chosen messages to random ones~\cite{GHHM21}. This remains a conjecture: the first bound is proved for the WOTS+ encoding rather than for a target sum, and the second reduction induces a small loss of around 1.5 bits.

\section{Completeness}

\begin{definition}[Completeness]
With $\hash$ modeled as a random oracle, the completeness error of $\SIG$ is the worst-case probability that the signer fails,
\[
  \varepsilon_{\SIG}=\max_{\ep,\,m}\ \Pr\!\left[\sigma=\bot\ \middle|\ (\pk,\sk)\gets\Gen,\ \sigma\gets\Sig(\sk,\ep,m)\right],
\]
over the oracle and the randomness of $\Gen$ and $\Sig$. The scheme is $\varepsilon$-complete if $\varepsilon_{\SIG}\leq\varepsilon$.
\end{definition}

\begin{theorem}[XMSS completeness]
The XMSS specified above is $2^{-256}$-complete in the ROM. Consequently a single key answers all $L=2^{32}$ of its epochs with a signature, except with probability at most $2^{-224}$.
\end{theorem}

\begin{proof}
Fix $\ep$ and $m$. As $\ep$ is unused, $\Sig(\sk,\ep,m)$ returns $\bot$ only by exhausting its $\amax$ attempts. Each attempt makes one oracle query,
\[
  I(\rho)=\mathsf{tweak}_{\mathrm{enc}}(\ep)\concat P\concat m\concat\rho\concat0^{64},
  \qquad D=\Truncate_{n\text{ bits}}\!\left(\hash(I(\rho))\right),
\]
and succeeds or not according to $D$ alone: it succeeds exactly when $D\in\mathcal D$, where
\[
  \mathcal D=\left\{D\in\bits n:\text{bit }63\text{ of }d_0\text{ and of }d_1\text{ is zero, and }x\in\mathcal C\right\}
\]
is a fixed subset of $\bits n$, reading $d_0$, $d_1$ and $x$ off $D$ as in $\IncEnc$. Put $\alpha=|\mathcal D|/2^n$.

\emph{The count.} As $wv+2=n$, the digits of $x$ and the two padding bits partition the bit positions of $D$, so zeroing the padding bits puts $\mathcal D$ in bijection with the $x\in\{0,\ldots,2^w-1\}^v$ summing to $T$. Counting those is classical~\cite{Comtet74}:
\[
  |\mathcal D|=\sum_{b=0}^{\lfloor T/2^w\rfloor}(-1)^b\binom{v}{b}\binom{T-b2^w+v-1}{v-1}.
\]
At $w=3$, $v=42$ and $T=195$,
\[
  |\mathcal D|=11539185377238682781344003244544752,\qquad \alpha=2^{-14.8479\ldots}\geq2^{-15}+2^{-128}.
\]

\emph{One attempt.} The map $I$ is injective, and no query of $\Gen$ reaches its image, $\Gen$ using tweak types $0$, $1$, $2$, $3$ and $10$, not $4$. So a $\rho$ not drawn before is a fresh oracle point, making $D$ uniform on $\bits n$ and independent of the history, and the attempt succeeds with probability exactly $\alpha$.

\emph{Exhausting the attempts.} Let $\mathcal F_a$ be the event that the first $a$ attempts all fail. The signer then samples a new $\rho$, uniformly at random and independently of the earlier attempts. This $\rho$ equals one of the $a$ earlier ones with probability at most $a2^{-\lrnd}$, and otherwise the attempt fails with probability $1-\alpha$, so $\Pr[\mathcal F_{a+1}\mid\mathcal F_a]\leq1-\alpha+a2^{-\lrnd}$. Multiplying over $0\leq a<\amax$, and since $\amax2^{-\lrnd}=2^{-169}\leq2^{-128}$,
\[
  \Pr[\sigma=\bot]=\Pr[\mathcal F_{\amax}]\leq\left(1-\alpha+\amax2^{-\lrnd}\right)^{\amax}\leq\left(1-2^{-15}\right)^{\amax}.
\]
As $\left(1-2^{-15}\right)^{2^{15}}\leq2^{-1}$ and $\amax=2^{15}\cdot2^{8}$, this gives $\varepsilon_{\SIG}\leq2^{-2^{8}}=2^{-256}$. Finally $\mathsf{tweak}_{\mathrm{enc}}(\ep)$ carries the epoch, so calls at distinct epochs query disjoint oracle points and the bound holds for each of a key's $L$ epochs; a union bound over them gives the second claim.
\end{proof}
