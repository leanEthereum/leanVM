\title{A SPHINCS$^+$ variant}
\author{}
\date{}

\begin{document}
\maketitle

\begin{abstract}

This specification defines a stateless SPHINCS$^+$ variant for up to $2^{24}$ signing requests per key pair:

\begin{itemize}
   \item \textbf{NIST security level~1}~\cite{NISTPQC}, $\approx 128$-bit (resp. $\approx 64$-bit) of classical (resp. quantum) security, in the Random Oracle model, ROM (resp. Quantum Random Oracle model, QROM).\footnote{The classical bound is proved (Section~\ref{sec:security}); the quantum one is a target, not a proved statement (Section~\ref{sec:quantum}).}
  \item \textbf{Public key: 32 bytes.}
  \item \textbf{Signature: 4924 bytes.}
  \item \textbf{Verification: 497 hashes.}
  \item \textbf{Key generation: 1.38M hashes.}
  \item \textbf{Signing: approximately 190K hashes on average}, with a 1024-byte public cache.
\end{itemize}
\end{abstract}

The construction uses WOTS$^+$C and FORS$^+$C~\cite{HK22C,KN25} within the SPHINCS$^+$ framework~\cite{SPHINCSPLUS,FIPS205}.

\section{Definitions and notation}

\begin{definition}[Signature scheme]
A signature scheme is a tuple $\SIG=(\Gen,\Sig,\Ver)$, where $\Gen$ and $\Sig$ are randomized and $\Ver$ is deterministic:
\[
  \Gen\longrightarrow(\pk,\sk),\qquad
  \Sig(\sk,m)\longrightarrow\sigma\in\Sigma\cup\{\bot\},\qquad
  \Ver(\pk,m,\sigma)\longrightarrow\{0,1\},
\]
where $\Sigma$ is the signature space and $m\in\bits{\lmsg}$. Correctness requires $\Ver(\pk,m,\sigma)=1$ whenever $(\pk,\sk)$ is generated by $\Gen$ and $\Sig(\sk,m)$ returns $\sigma\neq\bot$. Signing is stateless and permits repeated messages.
\end{definition}

Byte strings are concatenated with $\concat$. Bits and integer encodings are little endian. $\LE_r(a)$ is the unsigned $r$-bit encoding of $a$. All indices are zero based. Layers are numbered from the top: layer $0$ carries the public key, layer $d-1$ signs few-time keys.

Let $\hash:\bits{*}\to\bits{256}$ be a cryptographic hash function, $\Truncate_\nu$ keep its first $\nu$ output bits, and $\mathcal H=\bits{n}$. For $P\in\bits{\lpar}$, $\tw\in\bits{\ltwk}$ and $M\in\bits{*}$, define
\[
  \Th(P,\tw,M)=\Truncate_n\!\left(\hash(\tw\concat P\concat M)\right).
\]

\begin{definition}[Tweak encoding]
Let $\mathsf{protocol\_domain\_sep}=1$. For one-byte $t$ and $\lay$, and unsigned 32-bit integers $\tau$, $p$ and $j$, define the 16-byte tweak
\[
\begin{aligned}
  \mathsf{enc}(t,\lay,\tau,p,j)={}&\LE_8(\mathsf{protocol\_domain\_sep})\concat\LE_8(t)\concat\LE_8(\lay)\concat\LE_8(0)\\
  &{}\concat\LE_{32}(p)\concat\LE_{32}(\tau)\concat\LE_{32}(j).
\end{aligned}
\]
\end{definition}

Use the following tweaks, where $(\lay,\tau)$ identifies a hypertree layer and tree, and $(\idx,\kappa)$ a few-time instance and tree:
\[
\begin{aligned}
  \mathsf{tw}_{\mathrm{prf}}(\lay,\tau,i,e) &= \mathsf{enc}(0,\lay,\tau,i,e),\\
  \mathsf{tw}_{\mathrm{chain}}(\lay,\tau,e,i,\mu) &= \mathsf{enc}(1,\lay,\tau,2^wi+\mu-1,e), &&0\leq i<v,\ 0<\mu<2^w,\\
  \mathsf{tw}_{\mathrm{leaf}}(\lay,\tau,e) &= \mathsf{enc}(2,\lay,\tau,0,e),\\
  \mathsf{tw}_{\mathrm{node}}(\lay,\tau,\lambda,j) &= \mathsf{enc}(3,\lay,\tau,\lambda,j), &&1\leq\lambda\leq h_\lay,\ j<2^{h_\lay-\lambda},\\
  \mathsf{tw}_{\mathrm{enc}}(\lay,\tau,e) &= \mathsf{enc}(4,\lay,\tau,0,e),\\
  \mathsf{tw}_{\mathrm{ftsprf}}(\idx,\kappa,j) &= \mathsf{enc}(5,\kappa,\idx,0,j),\\
  \mathsf{tw}_{\mathrm{ftsleaf}}(\idx,\kappa,j) &= \mathsf{enc}(6,\kappa,\idx,0,j),\\
  \mathsf{tw}_{\mathrm{ftsnode}}(\idx,\kappa,\lambda,j) &= \mathsf{enc}(7,\kappa,\idx,\lambda,j), &&1\leq\lambda\leq a,\ j<2^{a-\lambda},\\
  \mathsf{tw}_{\mathrm{ftsroots}}(\idx) &= \mathsf{enc}(8,0,\idx,0,0),\\
  \mathsf{tw}_{\mathrm{msg}} &= \mathsf{enc}(9,0,0,0,0),\\
  \mathsf{tw}_{\mathrm{parameter}} &= \mathsf{enc}(10,0,0,0,0).
\end{aligned}
\]
Types $0$, $5$ and $10$ are used only for seed derivation (Remark~\ref{rem:seed}).

\section{The index}

An index $\idx\in\{0,\ldots,2^h-1\}$ selects a few-time key and a tree and leaf at each layer (Figure~\ref{fig:hypertree}):
\[
  \tau_\lay(\idx)=\left\lfloor\idx\,/\,2^{\sum_{j\geq\lay}h_j}\right\rfloor,
  \qquad
  e_\lay(\idx)=\left\lfloor\idx\,/\,2^{\sum_{j>\lay}h_j}\right\rfloor\bmod 2^{h_\lay}.
\]
For $1\leq\lay<d$,
\[
  \tau_\lay=\tau_{\lay-1}\cdot2^{h_{\lay-1}}+e_{\lay-1},
\]
so the one-time key at $(\lay-1,\tau_{\lay-1},e_{\lay-1})$ signs the root of tree $(\lay,\tau_\lay)$. The $2^h$ indices correspond one-to-one to the bottom-layer leaves.

\section{The one-time signature}
\label{sec:ots}

Each WOTS key at $(\lay,\tau,e)$ has $v$ chains of $2^w-1$ hash steps, with positions indexed by $w$-bit integers (Figure~\ref{fig:wots}). A signature reveals position $x_i$ of chain $i$; verification hashes forward to the endpoints. The codeword $x$ belongs to
\[
  \mathcal C=\left\{(x_0,\ldots,x_{v-1})\in\{0,\ldots,2^w - 1\}^{v}:\sum_{i=0}^{v-1}x_i=T\right\}.
\]
Equal sums prevent reaching a distinct codeword by increasing chain positions, replacing the Winternitz checksum with a sum check. The signer searches for an admissible encoding using a counter carried in the signature.

\begin{definition}[Encoding]
$\Enc(P,\lay,\tau,e,M,c)$ computes
\[
  D=\Th\!\left(P,\mathsf{tw}_{\mathrm{enc}}(\lay,\tau,e),M\concat\LE_{\lctr}(c)\right),
\]
writes $D=D_0\concat D_1$ with $D_0,D_1$ of $n/2$ bits, and lets $d_q$ be the integer encoded little endian by $D_q$. For $q\in\{0,1\}$ and $0\leq r<v/2$ it sets
\[
  x_{qv/2+r}=\left\lfloor\frac{d_q}{2^{wr}}\right\rfloor\bmod 2^w .
\]
It returns $x=(x_0,\ldots,x_{v-1})$ if bit $wv/2=63$ of both $d_0$ and $d_1$ is zero and $x\in\mathcal C$; otherwise $\bot$.
\end{definition}

The two fixed bits ensure that an admissible codeword determines all $128$ bits of $D$. A uniform digest is admissible with probability $p=|\mathcal C|/2^n\approx2^{-13.60}$.

\begin{definition}[Hash chain]
For a chain index $i$ at position $(\lay,\tau,e)$, a start $\mu$, a number of steps $s$ and a value $y\in\mathcal H$, define $\Chain_{\lay,\tau,e,i}(P,\mu,0,y)=y$ and, for $s\geq1$,
\[
  \Chain_{\lay,\tau,e,i}(P,\mu,s,y)=\Th\!\left(P,\mathsf{tw}_{\mathrm{chain}}(\lay,\tau,e,i,\mu+s),\Chain_{\lay,\tau,e,i}(P,\mu,s-1,y)\right),
\]
valid for $0\leq\mu<2^w$ and $0\leq s\leq2^w-1-\mu$.
\end{definition}

The secrets $\sk_{\lay,\tau,e,i}$ for $0\leq i<v$ are sampled by $\Gen$. Their chain endpoints and Merkle leaf are
\[
\begin{aligned}
  \pk_{\lay,\tau,e,i}&=\Chain_{\lay,\tau,e,i}\!\left(P,0,2^w-1,\sk_{\lay,\tau,e,i}\right),\\
  X^{\lay,\tau}_{0,e}&=\Th\!\left(P,\mathsf{tw}_{\mathrm{leaf}}(\lay,\tau,e),\pk_{\lay,\tau,e,0}\concat\cdots\concat\pk_{\lay,\tau,e,v-1}\right).
\end{aligned}
\]

\begin{definition}[$\OtsSign$]
$\OtsSign(P,\lay,\tau,e,M)$ tries counters $c=0,\ldots,\cmax-1$ in order until
\[
  x=\Enc(P,\lay,\tau,e,M,c)\neq\bot.
\]
It returns $\bot$ if none succeeds; otherwise it outputs $(c,\sigma)$ with
\[
  \sigma_i=\Chain_{\lay,\tau,e,i}\!\left(P,0,x_i,\sk_{\lay,\tau,e,i}\right),\qquad 0\leq i<v.
\]
\end{definition}

\begin{definition}[$\OtsLeaf$]
$\OtsLeaf(P,\lay,\tau,e,M,c,\sigma)$ computes $x=\Enc(P,\lay,\tau,e,M,c)$ and returns $\bot$ if it is $\bot$; otherwise it returns
\[
  \Th\!\left(P,\mathsf{tw}_{\mathrm{leaf}}(\lay,\tau,e),\pk'_0\concat\cdots\concat\pk'_{v-1}\right),
  \qquad
  \pk'_i=\Chain_{\lay,\tau,e,i}\!\left(P,x_i,2^w-1-x_i,\sigma_i\right).
\]
\end{definition}

\begin{remark}[One key, one message]
\label{rem:once}
Each one-time key is used only for a fixed message: a child-tree root on an upper layer, or a few-time public key on the bottom layer. Repeated use must also reveal the same codeword, which is why $\OtsSign$ always chooses the least admissible counter. Revealing two distinct codewords $x$ and $x'$ would expose the chain values at $\min(x_i,x'_i)$, allowing an adversary to construct signatures for any codeword componentwise above that minimum.
\end{remark}

\section{A layer}
\label{sec:layer}

Layer $\lay$'s tree $\tau$ is the Merkle tree over the $2^{h_\lay}$ one-time leaves $X^{\lay,\tau}_{0,e}$:
\[
  X^{\lay,\tau}_{\lambda,j}=\Th\!\left(P,\mathsf{tw}_{\mathrm{node}}(\lay,\tau,\lambda,j),X^{\lay,\tau}_{\lambda-1,2j}\concat X^{\lay,\tau}_{\lambda-1,2j+1}\right),
  \qquad 1\leq\lambda\leq h_\lay,\ 0\leq j<2^{h_\lay-\lambda}.
\]

\begin{definition}[$\TreeRoot$, $\TreePath$]
$\TreeRoot(P,\lay,\tau)=X^{\lay,\tau}_{h_\lay,0}$, and $\TreePath(P,\lay,\tau,e)=(A_0,\ldots,A_{h_\lay-1})$ with
\[
  A_\lambda=X^{\lay,\tau}_{\lambda,\lfloor e/2^\lambda\rfloor\oplus1}.
\]
\end{definition}

\begin{definition}[$\TreeFold$]
$\TreeFold(P,\lay,\tau,e,X,A)$ sets $V_0=X$ and, for $0\leq\lambda<h_\lay$ with $j=\lfloor e/2^{\lambda+1}\rfloor$,
\[
  V_{\lambda+1}=
  \begin{cases}
    \Th\!\left(P,\mathsf{tw}_{\mathrm{node}}(\lay,\tau,\lambda+1,j),V_\lambda\concat A_\lambda\right) & \text{if bit $\lambda$ of $e$ is }0,\\[2pt]
    \Th\!\left(P,\mathsf{tw}_{\mathrm{node}}(\lay,\tau,\lambda+1,j),A_\lambda\concat V_\lambda\right) & \text{otherwise,}
  \end{cases}
\]
and returns $V_{h_\lay}$.
\end{definition}

\section{The few-time signature}
\label{sec:fts}

A few-time key has $k-1$ Merkle trees of $2^a$ secrets. A signature opens one secret per tree at indices selected by the message digest (Figure~\ref{fig:fts}). An adversary can combine previous openings whenever all selected secrets have been revealed. FORS$^+$C omits the last of $k$ trees by requiring the last digest index to be zero.

The secrets $s_{\idx,\kappa,j}$ for $0\leq\kappa<k-1$ and $0\leq j<2^a$ are sampled by $\Gen$. Their trees are
\[
\begin{aligned}
  Y^{\idx,\kappa}_{0,j} &= \Th\!\left(P,\mathsf{tw}_{\mathrm{ftsleaf}}(\idx,\kappa,j),s_{\idx,\kappa,j}\right),\\
  Y^{\idx,\kappa}_{\lambda,j} &= \Th\!\left(P,\mathsf{tw}_{\mathrm{ftsnode}}(\idx,\kappa,\lambda,j),Y^{\idx,\kappa}_{\lambda-1,2j}\concat Y^{\idx,\kappa}_{\lambda-1,2j+1}\right), &&1\leq\lambda\leq a.
\end{aligned}
\]

\begin{definition}[$\FtsKey$]
$\FtsKey(P,\idx)=\Th\!\left(P,\mathsf{tw}_{\mathrm{ftsroots}}(\idx),Y^{\idx,0}_{a,0}\concat\cdots\concat Y^{\idx,k-2}_{a,0}\right)$.
\end{definition}

\begin{definition}[$\FtsOpen$]
$\FtsOpen(P,\idx,u)$, for indices $u=(u_0,\ldots,u_{k-1})$, outputs $(s,B)$ with, for $0\leq\kappa<k-1$,
\[
  s_\kappa=s_{\idx,\kappa,u_\kappa},
  \qquad
  B_{\kappa,\lambda}=Y^{\idx,\kappa}_{\lambda,\lfloor u_\kappa/2^\lambda\rfloor\oplus1},\qquad 0\leq\lambda<a.
\]
\end{definition}

\begin{definition}[$\FtsRec$]
$\FtsRec(P,\idx,u,s,B)$ sets, for $0\leq\kappa<k-1$, $W_{\kappa,0}=\Th(P,\mathsf{tw}_{\mathrm{ftsleaf}}(\idx,\kappa,u_\kappa),s_\kappa)$ and, for $0\leq\lambda<a$ with $j=\lfloor u_\kappa/2^{\lambda+1}\rfloor$,
\[
  W_{\kappa,\lambda+1}=
  \begin{cases}
    \Th\!\left(P,\mathsf{tw}_{\mathrm{ftsnode}}(\idx,\kappa,\lambda+1,j),W_{\kappa,\lambda}\concat B_{\kappa,\lambda}\right) & \text{if bit $\lambda$ of $u_\kappa$ is }0,\\[2pt]
    \Th\!\left(P,\mathsf{tw}_{\mathrm{ftsnode}}(\idx,\kappa,\lambda+1,j),B_{\kappa,\lambda}\concat W_{\kappa,\lambda}\right) & \text{otherwise,}
  \end{cases}
\]
and returns $\Th\!\left(P,\mathsf{tw}_{\mathrm{ftsroots}}(\idx),W_{0,a}\concat\cdots\concat W_{k-2,a}\right)$.
\end{definition}

\begin{definition}[Message digest]
$\Digest(P,\rootnode,m,\rho)$ returns
\[
  \Truncate_{h+ka}\!\left(\hash\!\left(\mathsf{tw}_{\mathrm{msg}}\concat P\concat \rho\concat\rootnode\concat m\right)\right).
\]
This is $h+ka=176$ bits, or $22$ bytes. Interpreting it as a little-endian integer $N$, define
\[
  \idx=N\bmod 2^{h},\qquad
  u_\kappa=\left\lfloor N\,/\,2^{h+\kappa a}\right\rfloor\bmod 2^{a},\qquad 0\leq\kappa<k,
\]
and the digest is admissible if $u_{k-1}=0$.
\end{definition}

\begin{remark}[The randomizer]
\label{rem:rho}
Sampling $\rho$ randomizes the digest after the adversary chooses the message. For a uniform digest, conditioning on $u_{k-1}=0$ leaves $\idx$ and the other leaf indices uniform because they occupy disjoint bits.
\end{remark}

\section{Key generation}

$\Gen$ independently samples $P\getsr\bits{\lpar}$, the one-time secrets $\sk_{\lay,\tau,e,i}\getsr\mathcal H$ for every layer, tree, leaf and chain, and the few-time secrets $s_{\idx,\kappa,j}\getsr\mathcal H$ for every instance, tree and leaf. It returns
\[
  \pk=(\rootnode,P),\qquad \rootnode=\TreeRoot(P,0,0),\qquad
  \sk=\left(P,\rootnode,(\sk_{\lay,\tau,e,i}),(s_{\idx,\kappa,j})\right).
\]
Serialize the public key as $\rootnode\concat P$, 32 bytes.

\begin{remark}[Seed derivation]
\label{rem:seed}
Sample a master secret $S\getsr\bits{256}$. Set $P=\Th(0^{\lpar},\mathsf{tw}_{\mathrm{parameter}},S)$ and derive the signing secrets as
\[
  \sk_{\lay,\tau,e,i}=\Th\!\left(P,\mathsf{tw}_{\mathrm{prf}}(\lay,\tau,i,e),S\right),
  \qquad
  s_{\idx,\kappa,j}=\Th\!\left(P,\mathsf{tw}_{\mathrm{ftsprf}}(\idx,\kappa,j),S\right).
\]
This gives $\sk=(P,\rootnode,S)$. Store $P\concat S$, 48 bytes; the root is recomputed from them. The formal theorem assumes independent secrets; the costs in Section~\ref{sec:costs} assume this derivation.
\end{remark}

\begin{remark}[Signer cache]
\label{rem:cache}
Caching the top tree's $2^6$ nodes at height $6$ takes 1024 bytes. Each signature then rebuilds one $2^6$-leaf subtree and at most $2^6-1$ nodes above it. These public nodes can be recomputed from the secret key.
\end{remark}

\section{Signing}
\label{sec:sign}

$\Sig(\sk,m)$:
\begin{enumerate}[leftmargin=2em]
\item Try at most $\amax$ independent samples $\rho\getsr\bits{\lrnd}$. Stop at the first admissible $\Digest(P,\rootnode,m,\rho)$, yielding $\idx$ and $u$. Return $\bot$ if all attempts fail.
\item $(s,B)\gets\FtsOpen(P,\idx,u)$ and $M_{d-1}\gets\FtsKey(P,\idx)$.
\item For $\lay=d-1$ down to $0$, set $\tau=\tau_\lay(\idx)$ and $e=e_\lay(\idx)$. Compute $(c_\lay,\sigma_\lay)\gets\OtsSign(P,\lay,\tau,e,M_\lay)$ and return $\bot$ immediately if it fails. Then compute
\[
\begin{aligned}
  A_\lay&\gets\TreePath(P,\lay,\tau,e),\\
  M_{\lay-1}&\gets\TreeRoot(P,\lay,\tau).
\end{aligned}
\]
\item Output $\sigma=\left(\rho,(s_\kappa,B_\kappa)_{\kappa<k-1},(c_\lay,\sigma_\lay,A_\lay)_{\lay<d}\right)$.
\end{enumerate}
$M_{-1}=\rootnode$ is computed and discarded.

Serialize $\rho$ first, then the few-time openings in increasing $\kappa$, then the layers in increasing $\lay$. Each opening contains its secret followed by its path. Each layer contains $\LE_{\lctr}(c_\lay)$, its chain values, then its path. Chain values and path nodes are ordered by increasing index. The total size is
\[
  \underbrace{16}_{\rho}+\underbrace{(k-1)(1+a)\cdot 16}_{\text{few-time}}+\underbrace{d\,(4+v\cdot 16)}_{\text{one-time}}+\underbrace{h\cdot 16}_{\text{paths}}
  =16+2464+2028+416=4924\text{ bytes.}
\]

\section{Verification}
\label{sec:ver}

$\Ver(\pk,m,\bot)=0$. Otherwise, parse $\pk=(\rootnode,P)$ and $\sigma$ as above, rejecting malformed inputs, then:
\begin{enumerate}[leftmargin=2em]
\item Compute $\Digest(P,\rootnode,m,\rho)$, giving $\idx$ and $u$, and reject unless $u_{k-1}=0$.
\item $M_{d-1}\gets\FtsRec(P,\idx,u,s,B)$.
\item For $\lay=d-1$ down to $0$, set $\tau=\tau_\lay(\idx)$ and $e=e_\lay(\idx)$. Compute $X\gets\OtsLeaf(P,\lay,\tau,e,M_\lay,c_\lay,\sigma_\lay)$ and reject immediately if $X=\bot$. Then compute
\[
  M_{\lay-1}\gets\TreeFold(P,\lay,\tau,e,X,A_\lay).
\]
\item Accept if and only if $M_{-1}=\rootnode$.
\end{enumerate}

\label{sec:costs}

\section{Security}
\label{sec:security}

\subsection{Classical security}

\begin{definition}[Strong unforgeability in the ROM]
Let $\SIG=(\Gen,\Sig,\Ver)$ use a shared random oracle $\hash:\bits{*}\to\bits{256}$: each new input receives an independent uniform output, and repeated inputs receive the same output. The challenger runs $(\pk,\sk)\gets\Gen$ and gives $\pk$ to an arbitrary classical probabilistic adversary $\mathcal A$, with no time or memory bound. The adversary may adaptively:
\begin{enumerate}[leftmargin=2em]
  \item Query the random oracle on any input and receive its 256-bit output.
  \item Request a signature on $m\in\bits{\lmsg}$ and receive $\Sig(\sk,m)$, possibly $\bot$. At most $\qs$ requests are allowed, including repetitions and failures.
  \item Terminate with a claimed forgery $(m^*,\sigma^*)$.
\end{enumerate}
The adversary wins if $\Ver(\pk,m^*,\sigma^*)=1$ and no signing request for $m^*$ returned $\sigma^*$.

Call $\mathcal A$ $q$-bounded if every execution of this experiment makes at most $q$ hash calls. Count all calls, including repeated inputs and calls by key generation, the adversary, signing, and final verification. The scheme has $x$ bits of classical strong unforgeability in the ROM at $\qs$ signing requests if every $q\geq1$ and every $q$-bounded $\mathcal A$ satisfy
\[
  \Pr[\mathcal A\text{ wins}]\leq\frac{q}{2^{x}}.
\]
\end{definition}

\begin{theorem}[SPHINCS security]
The ideal SPHINCS scheme specified above, with independently sampled secrets, has 127 bits of classical strong unforgeability in the ROM at $\qs=2^{24}$ signing requests.
\end{theorem}

A formal Lean 4 proof using VCVio \cite{VCVio} is in \texttt{./formal/sphincs/}. The claim is stated in \texttt{SphincsSecurity/Statement.lean} and proved in \texttt{SphincsSecurity.lean}. The theorem depends only on Lean's three standard axioms.
