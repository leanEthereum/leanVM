\section{Our PCS: WHIR \slash Ligerito with affine folds}\label{annex:pcs}

A self-contained description of the WHIR\slash Ligerito polynomial commitment scheme (PCS) \cite{ACFY25,NA25} over binary fields, in the Johnson list-decoding regime.

A fold round combines two halves with the \emph{affine combiner} of the brilliant~\cite{CYWZ26}: the coefficient is a random polynomial $\afa+\afb X$ of degree one, not a random scalar as in the original schemes. Its mutual correlated agreement error is of order $1/|\E|$ instead of $\blen/|\E|$ for block length $\blen$ (Theorem~\ref{thm:mca-affine}, \S\ref{sec:affine-intuition}), so $128$-bit security is now possible with a $128$-bit field. We target 125 bits of security for practical reasons (due to grand product GKR elsewhere in the protocol).

%=====================================================================
\subsection{Preliminaries}\label{sec:prelims}
%=====================================================================

\subsubsection{Codes, interleaving, and list decoding}\label{sec:codes}\label{sec:binary-rs}

The encoder $\Enc_{\kappa,\rexp}$ of Definition~\ref{def:enc}, with $\rexp\ge1$ and $\kappa+\rexp\le64$, reads a message $g:\cube{\kappa}\to\E$ as the novel-basis polynomial $\mpoly_g$, of coefficient $g(u)$ on $\novel_{\langle u\rangle}$, and evaluates it on an additive domain $\dom$ of size $\blen=2^{\kappa+\rexp}$. It is a linear injection onto $\RS[\E,\dom,2^\kappa]$ (Definition~\ref{def:rs-code}) of rate $\rate=2^{-\rexp}$. The verifier uses its \emph{column weights}: for $x\in\dom$, $\Enc_{\kappa,\rexp}(g)[x]=\inner{W_x}{g}$ for every $g$, where the MLE $\mle{W_x}$ can be evaluated in $O(\kappa)$ field operations anywhere in $\E^\kappa$ (Lemma~\ref{lem:colweight}). A fold round raises degrees by one, so the analysis also uses polynomials of degree at most some $\dgr\ge2^\kappa$.

\begin{definition}[Interleaved words, lists]\label{def:interleaved}
An \emph{interleaved word} is a matrix $\orc \in \E^{\cube{\ell} \times \dom}$, of $2^\ell$ \emph{lanes}. A \emph{polynomial tuple} $U=(U_u)_{u\in\cube{\ell}}$ has degree at most $\dgr$ if every $U_u$ does; it \emph{agrees} with $\orc$ on $\aset\subseteq\dom$ if $U_u(x)=\orc[u,x]$ for all $u$ and all $x\in\aset$. For an agreement $\agr$, $\lst(\orc;\agr,\dgr)$ is the set of polynomial tuples of degree at most $\dgr$ agreeing with $\orc$ on at least $\agr$ coordinates. A tuple \emph{decodes} to the message $g_U:\cube{\ell+\kappa}\to\E$, where $g_U(u,\cdot)$ are the coefficients of $U_u$ below $2^\kappa$, its \emph{low part}; tuples of degree below $2^\kappa$ decode to distinct messages.
\end{definition}

\begin{definition}[Weighted claims on tuples]\label{def:tuple-claim}
For a weight $W:\cube{\ell}\times\cube{\kappa}\to\E$ and a polynomial tuple $P$, put $\inner{W}{P}:=\inner{W}{g_P}$ (Definition~\ref{def:inner}): a claim weighs only the low part, and coefficients of index $2^\kappa$ and above carry no weight.
\end{definition}

\begin{definition}[Affine fold]\label{def:affine-fold}
Let $\orc$ have $\ell\ge1$ lane variables, split a lane index as $(\iota,u')$, with $\iota\in\{0,1\}$ the variable folded first, and call $\orc_\iota:=\orc[(\iota,\cdot),\cdot]$ the two \emph{halves}. For $\afa,\afb\in\E$, the \emph{affine fold} is the interleaved word with $\ell-1$ lane variables whose column at $x\in\dom$ is
\[
\orc_{\afa,\afb}[u',x]\;:=\;\orc_1[u',x]+(\afa+\afb x)\,\orc_0[u',x].
\]
The same formula, with the indeterminate $X$ for $x$, folds a polynomial tuple $U$ into $U_{\afa,\afb}$, of degree at most one more. Evaluation commutes with the fold, so a tuple agreeing with $\orc$ on $\aset$ folds to one agreeing with $\orc_{\afa,\afb}$ on $\aset$.
\end{definition}

In a fold round, $F:=U_{(0,\cdot)}$ and $G:=U_{(1,\cdot)}$ are the halves whose folded bit is $0$ and $1$, and the fold is $G+(\afa+\afb X)F$.

\begin{theorem}[Johnson bound]\label{thm:johnson-general}
Let $\mathcal{C} \subseteq \Sigma^\blen$ be a code, over any alphabet $\Sigma$, in which two distinct codewords agree on at most $\dgr$ coordinates, and let $\agr$ be an integer with $\agr^2>\blen\dgr$. Then every word $\mathbf{c} \in \Sigma^\blen$ has at most $\jl(\agr,\dgr):=\bigl\lfloor\blen(\agr-\dgr)/(\agr^2-\blen\dgr)\bigr\rfloor$ codewords agreeing with it on at least $\agr$ coordinates.
\end{theorem}

\begin{proof}
Let $U_1, \dots, U_L$ be distinct codewords, each agreeing with $\mathbf{c}$ on at least $\agr$ coordinates, and fix for each $j$ a set $\aset_j$ of exactly $\agr$ such coordinates. For a coordinate $x$ let $N_x := |\{ j : x \in \aset_j \}|$, so that $\sum_x N_x = L \agr$. Two codewords that both agree with $\mathbf{c}$ at $x$ agree with each other there, so $|\aset_j \cap \aset_{j'}| \le \dgr$ for $j \neq j'$, and counting ordered pairs,
\[
\sum_x N_x (N_x - 1) \;=\; \sum_{j \neq j'} |\aset_j \cap \aset_{j'}| \;\le\; L(L-1)\, \dgr .
\]
By Cauchy-Schwarz, $\sum_x N_x^2 \ge \bigl(\sum_x N_x\bigr)^2/\blen = L^2\agr^2/\blen$, so the left side is at least $L^2\agr^2/\blen - L\agr$. Dividing by $L$ and multiplying by $\blen$ gives $L\agr^2-\blen\agr\le(L-1)\,\blen\dgr$, that is $L\,(\agr^2-\blen\dgr)\le\blen(\agr-\dgr)$. As $\agr^2>\blen\dgr$, $L\le\blen(\agr-\dgr)/(\agr^2-\blen\dgr)$, and $L$ is an integer.
\end{proof}

\begin{corollary}[Johnson bound for interleaved words]\label{thm:johnson-interleaved}
For every interleaved word $\orc$ over a domain of size $\blen$, every degree bound $\dgr<\blen$ and every agreement $\agr$ with $\agr^2>\blen\dgr$: $\ |\lst(\orc;\agr,\dgr)| \le \jl(\agr,\dgr)$.
\end{corollary}

\begin{proof}
Polynomial tuples of degree at most $\dgr$ are a code as in Theorem~\ref{thm:johnson-general} over $\Sigma=\E^{2^\ell}$: two distinct ones differ in some lane, and two distinct polynomials of degree at most $\dgr$ agree on at most $\dgr$ points.
\end{proof}

\subsection{The protocol}\label{sec:protocol}
%=====================================================================

\begin{protocol}[WHIR/Ligerito with affine folds]\label{fig:protocol}
\ \par\smallskip
\emph{Parameters.} The scheme commits to $f : \cube{\nv} \to \K$ across $\nlev \ge 1$ \emph{levels} $i = 0, \dots, \nlev-1$, with variable counts $\nv = \nv_0 > \nv_1 > \dots > \nv_\nlev \ge 0$, fold factors $\ell_i := \nv_i - \nv_{i+1}$ and $\nv_\nlev$ small (e.g.\ $\nv_\nlev \le 5$). Level $i$ encodes with $\Enc_i := \Enc_{\kappa_i, \rexp_i}$, $\kappa_i := \nv_{i+1}$, of rate $\rate_i = 2^{-\rexp_i}$ over a domain $\dom_i \subseteq \K$ of size $\blen_i = 2^{\kappa_i + \rexp_i}$ (requires $\kappa_i + \rexp_i \le 64$ and $\ell_i\le2^{\kappa_i}$), and takes $\nq_i$ queries. (The agreement thresholds appear only in the analysis.)

\smallskip
\textbf{Commit}$(f : \cube{\nv} \to \K)$: $\mathbf{P}$ sends the oracle $\orc^{(0)}[u, \cdot] := \Enc_0(f(u, \cdot))$, $u \in \cube{\ell_0}$, whose entries are $\K$ elements. Row $u$ evaluates on $\dom_0$ the \emph{lane} $P_u:=\mpoly_{f(u,\cdot)}$, the polynomial whose novel-basis coefficients are $f(u,\cdot)$, and the lanes form the polynomial tuple $P:=(P_u)_{u\in\cube{\ell_0}}$ (Definition~\ref{def:interleaved}), of message $g_P=f$. $\mathbf{V}$ sends $z_0\leftarrow\E^{\nv}$; $\mathbf{P}$ sends $\cood{0}\in\E$ (honestly $\mle{f}(z_0)$). The \emph{OOD claim} $(\eq(z_0,\cdot),\cood{0})$ joins the claims the opening proves.

\smallskip
\textbf{Open}$\bigl(\{(W_k, c_k)\}_{k = 1, \dots, J_0}\bigr)$, $1\le J_0\le512$ weighted claims about $f$: $\mathbf{V}$ sends $\lambda \leftarrow \E$. The running claim is $(W, \rc{0}) := \bigl(\sum_{k} \lambda^{k-1} W_k,\; \sum_{k} \lambda^{k-1} c_k\bigr)$, a weighted claim (Definition~\ref{def:tuple-claim}) about the lanes $P$, which the prover folds below. For $i = 0, 1, \dots, \nlev - 1$, with $\kappa:=\kappa_i$:
\begin{enumerate}[itemsep=0.5pt,topsep=2pt]
\item \emph{(fold rounds)} For $j = 1, \dots, \ell_i$, with running claim $(W, \rc{j-1})$, $W:\cube{\ell_i-j+1}\times\cube{\kappa}\to\E$, and lanes
\[
\begin{gathered}
P=(P_u)_{u\in\cube{\ell_i-j+1}},\qquad P_u\in\E[X],\qquad \deg P_u<2^\kappa+j-1,\\
F:=(P_{(0,u')})_{u'\in\cube{\ell_i-j}},\qquad G:=(P_{(1,u')})_{u'\in\cube{\ell_i-j}}:
\end{gathered}
\]
\begin{enumerate}[itemsep=0.5pt,topsep=1pt,label=(\alph*)]
\item $\mathbf{P}$ sends $v_G, v_F, \tabx{0}, \tabx{1} \in \E$, honestly $v_G = \inner{W(0,\cdot)}{G}$, $v_F=\inner{W(0,\cdot)+W(1,\cdot)}{F}$ and $\tabx{\iota}=\inner{W(\iota,\cdot)}{XF}$, with $XF:=(X F_{u'})_{u'}$.
\item $\mathbf{V}$ sends $\afa_j, \afb_j \leftarrow \E$.
\item Both set $\rc{j} := v_G+\afa_j\bigl(v_G+\rc{j-1}\bigr)+\afa_j^2v_F+\afb_j\tabx{0}+\afa_j\afb_j\bigl(\tabx{0}+\tabx{1}\bigr)$ and $W \gets \mle{W}(\afa_j,\cdot)$; $\mathbf{P}$ sets $P \gets P_{\afa_j,\afb_j} = G+(\afa_j+\afb_jX)F$.
\end{enumerate}
After the $\ell_i$ rounds, $P$ is one polynomial, $w:=\orc^{(i)}_{\ell_i}\in\E^{\dom_i}$ is the oracle folded by the same challenges (Definition~\ref{def:affine-fold}; $\orc^{(i)}_j$ is $\orc^{(i)}$ folded by the first $j$ rounds), honestly $P$ on $\dom_i$, and
\[
P=\mpoly_{f^{(i+1)}}+\Wn{\kappa}\textstyle\sum_{j<\ell_i}\phigh_j\novel_j,\qquad \sum_{x\in\cube{\kappa}}W(x)\,f^{(i+1)}(x)=\rc{\ell_i}.
\]
The claim ignores the high part $\phigh$, as the low coefficients of $XF$ depend only on those of $F$ (Lemma~\ref{lem:mulx}(ii)); step 2(d) removes it from the queried values.
\item If $i < \nlev - 1$:
\begin{enumerate}[itemsep=0.5pt,topsep=1pt,label=(\alph*)]
\item \emph{(commit)} $\mathbf{P}$ sends the oracle $\orc^{(i+1)}[u, \cdot] := \Enc_{i+1}(f^{(i+1)}(u, \cdot))$, $u \in \cube{\ell_{i+1}}$, row $u$ evaluating the new lane $P_u:=\mpoly_{f^{(i+1)}(u,\cdot)}$, then $\phigh$.
\item \emph{(ood)} $\mathbf{V}$ sends $z_{i+1} \leftarrow \E^{\nv_{i+1}}$; $\mathbf{P}$ sends $\cood{i+1} \in \E$ (honestly $\mle{f^{(i+1)}}(z_{i+1})$).
\item \emph{(query)} $\mathbf{V}$ sends $\nq_i$ i.i.d.\ queries $x_q \leftarrow \dom_i$, $q<\nq_i$, and $\lambda\leftarrow\E$, one message.
\item \emph{(read)} $\mathbf{V}$ reads the columns $\orc^{(i)}[\cdot, x_q]$, folds each to $w(x_q)$, and removes the high part, $c_q := w(x_q) + \Wn{\kappa}(x_q)\sum_{j<\ell_i}\phigh_j\,\novel_j(x_q)$ (honestly $\Enc_i(f^{(i+1)})[x_q]$).
\item \emph{(next level)} The $J_{i+1}:=\nq_i+2$ claims of level $i+1$, the residual, the OOD and the $\nq_i$ consistency claims, are batched into one running claim about the new lanes:
\[
W \gets W + \lambda\,\eq(z_{i+1},\cdot) + \lambda^2\textstyle\sum_q \lambda^{q} W^{(i)}_{x_q},
\qquad
\rc{0} := \rc{\ell_i} + \lambda\,\cood{i+1} + \lambda^2\textstyle\sum_{q}\lambda^{q}c_q,
\]
where $W^{(i)}_x$ are the column weights of $\Enc_i$ (\S\ref{sec:codes}).
\end{enumerate}
\item If $i = \nlev - 1$: \emph{(final)} $\mathbf{P}$ sends the $2^{\nv_\nlev}+\ell_i$ coefficients of $P$. $\mathbf{V}$ sends $\nq_{\nlev-1}$ i.i.d.\ queries $x_q \leftarrow \dom_{\nlev-1}$, computes $w(x_q)$ as in 2(d) and checks $w(x_q)\qeq P(x_q)$. Then $\nv_\nlev$ sumcheck rounds of degree two (Fact~\ref{fact:sumcheck}) reduce $\sum_x W(x)\,f^{(\nlev)}(x)\qeq\rc{\ell_i}$ to a point $\fc\in\E^{\nv_\nlev}$, and $\mathbf{V}$ checks the last running claim against $\mle{W}(\fc)\cdot\mle{f^{(\nlev)}}(\fc)$, from the coefficients it holds.
\end{enumerate}
$\mathbf{V}$ evaluates $\mle{W}$ itself (MLE-friendly), at the point $(\afa,\fc)$ of all rounds' $\afa_j$ and tail challenges. $\mathbf{V}$ accepts iff every check above passed.
\end{protocol}

\subsection{Why an affine fold}\label{sec:affine-intuition}
%=====================================================================

Let $\orc_0,\orc_1:\dom\to\E$ be two words and $F,G$ two codewords, with errors $e_0:=\orc_0-F$ and $e_1:=\orc_1-G$. A point $x$ is \emph{bad} if $e_0(x)\neq0$ or $e_1(x)\neq0$, and a fold challenge \emph{repairs} it if the folded error vanishes at $x$.

\paragraph{Scalar fold.} The folded error of $\orc_1+\afa\,\orc_0$ is $e_1+\afa e_0$, so each bad point has at most one repairing challenge, $\afa=-e_1(x)/e_0(x)$, which the prover chooses through the words. If they agree with $F,G$ on $\agr-1$ common points and with no pair of codewords on $\agr$, the prover gives the other $\blen-\agr+1$ points distinct repairing challenges, and the fold agrees with $G+\afa F$ on $\agr$ points with probability $(\blen-\agr+1)/|\E|$; tolerating the loss of a few coordinates does not help \cite[Prop.~6.1]{CYWZ26}, after \cite[Remark~1.1]{BCIKS20}.

\paragraph{Affine fold.} The folded error of $\orc_1+(\afa+\afb x)\orc_0$ is $e_1+(\afa+\afb x)e_0$, so $(\afa,\afb)$ repairs $x$ exactly when $e_0(x)\neq0$ and $\afa+x\afb=-e_1(x)/e_0(x)$. These challenges form a line in $\E^2$ whose position the prover sets but whose slope $x$ sets, so the lines of two different points cross at exactly one challenge. A random challenge thus repairs one bad point with probability up to $(\blen-\agr+1)/|\E|$, but two with probability at most $\binom{\blen}{2}/|\E|^2$: tolerating the loss of one coordinate (Theorem~\ref{thm:mca-affine}) leaves an error of order $1/|\E|$.

\subsection{Round-by-round soundness}\label{sec:soundness}
%=====================================================================

\paragraph{Thresholds.} At level $i$, with $\blen:=\blen_i$ and $\ell:=\ell_i$, every round is analysed at the level's final degree $\dgr$, and round $j$, $0\le j\le\ell$, at the agreement $\agr_j$:
\[
\dgr := 2^{\kappa_i}-1+\ell,
\qquad
\agr_j := \agr_\ell-(\ell-j),
\]
where $\agr_\ell\le\blen$, the agreement at the level's last round, is a parameter. The honest lanes start at degree $2^{\kappa_i}-1$ and gain one degree per round, so they stay within $\dgr$; each round loses one coordinate of agreement. Write $\agr^{(i)}_j$, $\dgr^{(i)}$ when the level matters, and $L_i:=\jl(\agr^{(i)}_0,\dgr^{(i)})$, the size of the list the level starts on (Corollary~\ref{thm:johnson-interleaved}).

\begin{theorem}[RBR soundness]\label{thm:rbr}
Suppose that $(\agr^{(i)}_0)^2>\blen_i\dgr^{(i)}$ for every level $i$, so that the Johnson counts below are defined (as $\agr^{(i)}_j-1\ge\agr^{(i)}_0$ for $j\ge1$), and let $\epsaff$ be the affine MCA error of Definition~\ref{def:epsaff}. Call the input claims $\{(W_k, c_k)\}_k$ \emph{false} if no $U \in \lst(\orc^{(0)};\agr^{(0)}_0,\dgr^{(0)})$ has $\inner{W_k}{g_U} = c_k$ for all $k$. Then Protocol~\ref{fig:protocol} has the following RBR soundness error at each verifier message, with $\blen:=\blen_i$, $\agr_j:=\agr^{(i)}_j$, $\dgr:=\dgr^{(i)}$:
\[
\begin{array}{llcl}
\textnormal{batching } \lambda \textnormal{ of level } 0: & \varepsilon^{\mathrm{batch}} &=& \dfrac{(J_0 - 1)\, L_0}{|\E|} , \\[2ex]
\textnormal{fold } (\afa_j,\afb_j), \textnormal{ level } i: & \varepsilon^{\mathrm{fold}}_{i,j} &=& \epsaff(\blen,\dgr,\agr_j) + \dfrac{2\,\jl(\agr_j-1,\dgr)}{|\E|} , \\[2ex]
\textnormal{OOD point } z_i: & \varepsilon^{\mathrm{ood}}_i &=& \dbinom{L_i}{2} \dfrac{\nv_i}{|\E|} , \\[2ex]
\textnormal{query message of level } i: & \varepsilon^{\mathrm{query}}_i &=& \Bigl(\dfrac{\agr_{\ell_i}-1}{\blen}\Bigr)^{\nq_i} + \dfrac{(J_{i+1} - 1)\, L_{i+1}}{|\E|} , \\[2ex]
\textnormal{each of the } \nv_\nlev \textnormal{ tail rounds}: & \varepsilon^{\mathrm{tail}} &=& \dfrac{2}{|\E|} ,
\end{array}
\]
the second query term being absent at the last level, $i=\nlev-1$. Each row is divided by $2^{b}$ for the grind of $b$ bits before its message, which the protocol leaves implicit; the \emph{RBR error}, which governs the Fiat-Shamir compilation, is the largest result, held to $2^{-125}$ (\S\ref{sec:grinding}).
\end{theorem}

\subsubsection{Proximity tools}\label{sec:tools}

The main tool is the \emph{mutual correlated agreement} (MCA)~\cite{ACFY25} of the affine combiner, with the error bound of~\cite[Lemma~4.1]{CYWZ26}.

\begin{definition}[Affine MCA error]\label{def:epsaff}
Let $1\le\dgr<\agr\le\blen$ be integers. A triple $(\imul,\iyz,\ithr)$ of positive integers is \emph{admissible} if, with $\iz:=\min\bigl(\iyz,\lfloor(\imul\agr-\iyz-1)/\dgr\rfloor\bigr)$,
\begin{enumerate}[itemsep=1pt,label=(\roman*)]
\item $\iz\ge1$;
\item the \emph{interpolation inequality} holds:
\[
\sum_{j=0}^{\iz}\max(0,\imul\agr-\iyz-\dgr j)\,(\iyz-j+1)\;>\;\blen\sum_{s=0}^{\min(\imul-1,\iz)}(\imul-s)(\iyz-s+1);
\]
\item $\dgr+1<\ithr<\agr$ and $\ithr^2>\blen(\dgr+1)$.
\end{enumerate}
Then
\[
\epsaff(\blen,\dgr,\agr)\;:=\;\min_{(\imul,\iyz,\ithr)\ \mathrm{admissible}}\ \frac{\iyz^2+\jl(\ithr,\dgr+1)\cdot\max\bigl(|\E|,\binom{\blen}{2}\bigr)+4\dgr^2\iz^2\iyz^3\bigl(\blen/(\agr-\ithr)\bigr)^2}{|\E|^2}.
\]
\end{definition}

\begin{theorem}[Affine MCA; {\cite[Thm.~1.1, Lemma~4.1, \S5.1]{CYWZ26}}]\label{thm:mca-affine}
Let $\dom\subseteq\K$ have size $\blen$, let $1\le\dgr<\agr\le\blen$, $h\ge1$, and let $\orc_0,\orc_1:\dom\to\E^h$ be words of $h$ lanes. For $\afa,\afb\in\E$ and an $h$-tuple $P$ of polynomials let $\aset_{\afa,\afb}(P):=\{x\in\dom:P(x)=\orc_1(x)+(\afa+\afb x)\orc_0(x)\}$. Then, for uniform $\afa,\afb\leftarrow\E$, except with probability at most $\epsaff(\blen,\dgr,\agr)$: every $h$-tuple $P$ of degree at most $\dgr$ with $|\aset_{\afa,\afb}(P)|\ge\agr$ has $h$-tuples $G$, $F$ with $\deg G\le\dgr$, $\deg F\le\dgr-1$ and $P=G+(\afa+\afb X)F$, such that $G=\orc_1$ and $F=\orc_0$ on all but at most one point of $\aset_{\afa,\afb}(P)$ (agreement loss one).
\end{theorem}

For $h=1$ this is \cite[Thm.~1.1, Lemma~4.1]{CYWZ26}, which holds for every admissible triple; \cite[\S5.1]{CYWZ26} extends it to $h$ lanes, packed into one word over an extension of $\E$, with the same error.

\begin{lemma}[An affine fold round]\label{lem:fold-list}
Let $\orc$ be an interleaved word with $\ell\ge1$ lane variables over a domain of size $\blen$, with halves $\orc_0,\orc_1$ of $h=2^{\ell-1}$ lanes each. Let $1\le\dgr<\agr\le\blen$ with $(\agr-1)^2>\blen\dgr$, and $L:=\jl(\agr-1,\dgr)$. Let $W:\cube{\ell}\times\cube{\kappa}\to\E$ be a weight, $W_\iota:=W(\iota,\cdot)$, $c\in\E$, and suppose every tuple of $\lst(\orc;\agr-1,\dgr)$, as a pair $(F,G)$ of halves, has $\inner{W_0}{F}+\inner{W_1}{G}\neq c$. Let $v_G,v_F,\tabx{0},\tabx{1}\in\E$ be arbitrary. Except with probability at most $\epsaff(\blen,\dgr,\agr)+2L/|\E|$ over uniform $(\afa,\afb)$, every $P\in\lst(\orc_{\afa,\afb};\agr,\dgr)$ has
\[
\inner{\mle{W}(\afa,\cdot)}{P}\;\neq\;v_G+\afa(v_G+c)+\afa^2v_F+\afb\tabx{0}+\afa\afb(\tabx{0}+\tabx{1}).
\]
\end{lemma}

This is \cite[Thm.~5.1]{CYWZ26} without its premix challenge, which a degree fixed for the whole level makes unnecessary.

\begin{proof}
The list $\lst(\orc;\agr-1,\dgr)$ has at most $L$ members (Corollary~\ref{thm:johnson-interleaved}) and is fixed before the challenges. There are two bad events. $B_1$ is the event of Theorem~\ref{thm:mca-affine} for $\orc_0$ and $\orc_1$, of probability at most $\epsaff(\blen,\dgr,\agr)$. $B_2$ is that some pair $(F,G)$ of the list satisfies the new claim for $P_{F,G}:=G+(\afa+\afb X)F$. For a fixed pair, by the completeness argument, the true value $\inner{\mle{W}(\afa,\cdot)}{P_{F,G}}$ has the same form as the claimed one, with the pair's own values in place of $v_G,v_F,\tabx{0},\tabx{1}$ and $c^\ast:=\inner{W_0}{F}+\inner{W_1}{G}\neq c$ in place of $c$. The difference is a polynomial of degree at most $2$ in $(\afa,\afb)$. Its constant term and its coefficient of $\afa$ both vanish only if $c^\ast=c$, so it is nonzero and vanishes with probability at most $2/|\E|$ (Lemma~\ref{lem:sz}). A union bound over the list gives $\Pr[B_2]\le2L/|\E|$.

Outside both events, let $P\in\lst(\orc_{\afa,\afb};\agr,\dgr)$. By Theorem~\ref{thm:mca-affine}, $P=G+(\afa+\afb X)F$ with $\deg G\le\dgr$, $\deg F\le\dgr-1$, and $G=\orc_1$, $F=\orc_0$ on all but one of its agreement points. So $(F,G)$ is in $\lst(\orc;\agr-1,\dgr)$ and $P=P_{F,G}$, which violates the new claim as $B_2$ failed.
\end{proof}

\begin{lemma}[OOD separation]\label{lem:ood}
Let $\orc$ be an interleaved word with $|\lst(\orc;\agr,\dgr)| \le L$, decoding to $\nv$-variable messages. For uniform $z \leftarrow \E^{\nv}$, $\Pr[\exists\, U, U' \in \lst(\orc;\agr,\dgr) : g_U\neq g_{U'},\ \mle{g_U}(z) = \mle{g_{U'}}(z)] \le \binom{L}{2}\, \nv / |\E|$.
\end{lemma}

\begin{proof}
The list decodes to at most $L$ messages, and distinct messages have distinct multilinear extensions. By Lemma~\ref{lem:sz} two of them agree at $z$ with probability at most $\nv/|\E|$; a union bound over the pairs concludes.
\end{proof}

\subsubsection{Proof of the main theorem}\label{pcs:proof}

\begin{proof}[Proof of Theorem~\ref{thm:rbr}]
We say that a polynomial tuple $U$ \emph{violates} a claim $(W,c)$ if $\inner{W}{U}\neq c$. Since a claim only weighs the low part, whether $U$ violates it depends only on its message $g_U$. For a level $i$ and $0\le j\le\ell_i$, let
\[
\lst^{(i)}_j:=\lst\bigl(\orc^{(i)}_j;\,\agr^{(i)}_j,\,\dgr^{(i)}\bigr)
\]
be the list of the level-$i$ oracle folded by its first $j$ rounds. It has at most $L_i$ members.

\medskip\noindent\emph{The invariant.} At each point of the protocol, the invariant is that some check of $\mathbf{V}$ has already failed, or that the condition attached to the last verifier message holds:
\begin{itemize}[itemsep=1pt]
\item at the start of the opening: every $U\in\lst^{(0)}_0$ violates at least one input claim;
\item after the batching challenge of level $i$ ($j=0$) and after its $j$-th fold challenge:
\[
(\mathrm{I}_{i,j})\qquad\text{every } U\in\lst^{(i)}_j \text{ violates the running claim } (W,\rc{j});
\]
\item after the OOD point $z_{i+1}$: $(\mathrm{I}_{i,\ell_i})$ holds, and distinct messages $g$ of $\lst^{(i+1)}_0$ have distinct values $\mle{g}(z_{i+1})$;
\item after the final query message: $f^{(\nlev)}$ violates the running claim;
\item after each tail round: the running sumcheck claim is false.
\end{itemize}
Property (i) is the definition of false input claims. For (ii), each condition only involves messages sent up to the last verifier message, so the prover's next message cannot break it. For (iv), at the end either a check has failed, or the last sumcheck claim is false; $\mathbf{V}$ holds $f^{(\nlev)}$ and checks that claim itself, so it rejects. It remains to prove (iii): each verifier message breaks the invariant with at most the probability of the theorem.

\medskip\noindent\emph{Batching challenge $\lambda$.} Suppose every $U\in\lst^{(i)}_0$ violates at least one of $J$ claims $(W_k,c_k)$; at level $0$ this is the starting condition, with $J=J_0$. Fix such a $U$ and put $d_k:=\inner{W_k}{U}-c_k$. Then $U$ satisfies the batched claim only if $\sum_k\lambda^{k-1}d_k=0$. This is a nonzero polynomial in $\lambda$ of degree less than $J$, so by Lemma~\ref{lem:sz} it vanishes with probability at most $(J-1)/|\E|$. A union bound over the at most $L_i$ members gives $(\mathrm{I}_{i,0})$, except with probability $(J-1)L_i/|\E|$.

\medskip\noindent\emph{Fold challenge $(\afa_j,\afb_j)$.} Since $\agr^{(i)}_{j-1}=\agr^{(i)}_j-1$, the condition $(\mathrm{I}_{i,j-1})$ is the hypothesis of Lemma~\ref{lem:fold-list} with $\agr=\agr^{(i)}_j$, whatever four values the prover sent. The lemma gives $(\mathrm{I}_{i,j})$, except with probability $\epsaff(\blen_i,\dgr^{(i)},\agr^{(i)}_j)+2\jl(\agr^{(i)}_j-1,\dgr^{(i)})/|\E|$.

\medskip\noindent\emph{OOD point $z_{i+1}$.} The list $\lst^{(i+1)}_0$ is fixed by the prover's commitment, which comes before $z_{i+1}$. By Lemma~\ref{lem:ood}, its distinct messages have distinct values at $z_{i+1}$, except with probability $\binom{L_{i+1}}{2}\nv_{i+1}/|\E|$. The condition $(\mathrm{I}_{i,\ell_i})$ does not involve $z_{i+1}$, so it still holds. (The same bound at $z_0$ makes the commitment binding; the opening does not use it.)

\medskip\noindent\emph{Query message of level $i<\nlev-1$.} Before it, the prover has sent $\cood{i+1}$. By separation, at most one message $g^\ast$ of $\lst^{(i+1)}_0$ has $\mle{g^\ast}(z_{i+1})=\cood{i+1}$, and $g^\ast$ and the high part $\phigh$ are fixed before the queries. We first show that, except with probability $\bigl((\agr^{(i)}_{\ell_i}-1)/\blen_i\bigr)^{\nq_i}$, every $U\in\lst^{(i+1)}_0$ violates at least one of the $J_{i+1}$ claims of step 2(e).

Take $U\in\lst^{(i+1)}_0$. If $g_U\neq g^\ast$, $U$ violates the OOD claim. Otherwise it is enough to show that $g^\ast$ violates a claim. Let
\[
P^\ast:=\mpoly_{g^\ast}+\Wn{\kappa_i}\sum_{j<\ell_i}\phigh_j\novel_j,
\]
the polynomial of degree at most $\dgr^{(i)}$ with low part $g^\ast$ and high part $\phigh$.
\begin{itemize}[itemsep=1pt]
\item If $P^\ast$ agrees with $w$ on at least $\agr^{(i)}_{\ell_i}$ points, then $P^\ast$ violates $(W,\rc{\ell_i})$ by $(\mathrm{I}_{i,\ell_i})$. As this depends only on the low part, $g^\ast$ violates the residual claim.
\item Otherwise $P^\ast$ and $w$ agree on fewer than $\agr^{(i)}_{\ell_i}$ points, and all $\nq_i$ independent queries land on such points with probability at most $\bigl((\agr^{(i)}_{\ell_i}-1)/\blen_i\bigr)^{\nq_i}$. If some query has $P^\ast(x_q)\neq w(x_q)$, subtract $\Wn{\kappa_i}(x_q)\sum_{j<\ell_i}\phigh_j\novel_j(x_q)$ from both sides: the left side becomes $\mpoly_{g^\ast}(x_q)=\Enc_i(g^\ast)[x_q]$ and the right side becomes $c_q$. So $g^\ast$ violates the consistency claim $(W^{(i)}_{x_q},c_q)$.
\end{itemize}
The claims of step 2(e) are fixed by the queries, and $\lambda$ is uniform and independent of them. The batching argument then gives $(\mathrm{I}_{i+1,0})$, except with a further $(J_{i+1}-1)L_{i+1}/|\E|$.

\medskip\noindent\emph{Final query message.} At the last level $i=\nlev-1$, the prover has sent $P$ itself, a polynomial of degree at most $\dgr^{(i)}$ with low part $f^{(\nlev)}$. The two cases above apply to $P$ in place of $P^\ast$. If $P$ agrees with $w$ on at least $\agr^{(i)}_{\ell_i}$ points, $f^{(\nlev)}$ violates the running claim. Otherwise, except with probability $\bigl((\agr^{(i)}_{\ell_i}-1)/\blen_i\bigr)^{\nq_i}$, some query has $P(x_q)\neq w(x_q)$, and the check of $\mathbf{V}$ fails.

\medskip\noindent\emph{Tail round.} The tail is a sumcheck of degree two (Fact~\ref{fact:sumcheck}): if a round starts from a false claim, the claim after its challenge is false except with probability $2/|\E|$.
\end{proof}

\subsection{Optimizations}\label{sec:pcs-opt}
%=====================================================================

The committed $f$ is the stack of \S\ref{sec:stacking}: $\nstack$ elements of $\K$, then zeros up to $2^{\nv}$. Its row index is the stack's most significant $\ell_0$ coordinates, $f(u, x) := q(x, u)$ for $u \in \cube{\ell_0}$ and $x \in \cube{\nv - \ell_0}$, so row $u$ is the block of $2^{\nv - \ell_0}$ consecutive elements of $q$ at offset $\langle u \rangle 2^{\nv - \ell_0}$ (\S\ref{app:ntt}), the padding is whole rows, and only $\nrows=\lceil\nstack/2^{\nv-\ell_0}\rceil$ of the $2^{\ell_0}$ rows hold data. Claims on the stack, the commitment's OOD claim included, are taken in $q$'s variable order and rotated to $f$'s. At commitment:
\begin{itemize}[itemsep=1pt,topsep=2pt]
\item the encoder runs on the $\nrows$ data rows only, as encoding zeros gives zeros;
\item a Merkle leaf hashes all $2^{\ell_0}$ values of its column, but in reverse order, padding zeros first: the prover hashes the whole $136$-byte blocks of zeros once, shared by every leaf, and continues each leaf from that sponge state, and the proof stores only the $\nrows$ data values (\S\ref{sec:e2e-merkle-pruning}).
\end{itemize}
Both costs are thus proportional to $\nstack$, not to its next power of two. The padding inside the $\nstack$ elements, from instruction counts that are not powers of two, remains; the Jagged PCS \cite{jagged_pcs} would remove it.
