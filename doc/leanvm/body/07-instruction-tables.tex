\section{The instruction tables}\label{sec:tables}

Each instruction has its own table.

Write $\rdf{\sep,\addr,\cnt,\val_0,\val_1, \dots}$ for the pair \pull\ $\tup{\sep,\addr,\cnt,\val_0,\val_1, \dots }$, \push\ $\tup{\sep,\addr,\gen\cdot\cnt,\val_0,\val_1, \dots}$: every flush below but the state's.

\subsection{\texttt{XOR}}\label{sec:tab-xor}

\texttt{XOR}\ $[o_A,o_B,o_C]$ asserts $\loc{o_C}=\loc{o_A}+\loc{o_B}$, the $192$-bit field sum (in $\E$).
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_A,o_B,o_C$; the two read values $(v_{A,i})_{i\in\{0,1,2\}}$ and $(v_{B,i})_{i\in\{0,1,2\}}$; memory counts $r_A,r_B,r_C$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} $\rdf{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{XOR},o_A,o_B,o_C,0,0}$.
  \item \textbf{Memory:}
    \begin{align*}
      &\rdf{\dsep{MEM},\fp\cdot o_X,r_X,v_{X,0},v_{X,1},v_{X,2}}\quad(X\in\{A,B\}),\\
      &\rdf{\dsep{MEM},\fp\cdot o_C,r_C,v_{A,0}+v_{B,0},v_{A,1}+v_{B,1},v_{A,2}+v_{B,2}}.
    \end{align*}
  \end{itemize}
\end{itemize}

\subsection{\texttt{MUL\_NATIVE}}\label{sec:tab-mul}

\texttt{MUL\_NATIVE}\ $[o_A,o_B,o_C]$ asserts $\loc{o_C}=\loc{o_A} \cdot \loc{o_B}$, the $192$-bit field multiplication (in $\E$).
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_A,o_B,o_C$; the two read values $(v_{A,i})_{i\in\{0,1,2\}}$ and $(v_{B,i})_{i\in\{0,1,2\}}$; memory counts $r_A,r_B,r_C$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} $\rdf{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{MUL},o_A,o_B,o_C,0,0}$.
  \item \textbf{Memory} (the destination carries $v_Av_B$ in $\E$, of degree two). With $y^3=y+1$, put
    \[
    p_0=v_{A,0}v_{B,0},\quad
    p_1=v_{A,0}v_{B,1}+v_{A,1}v_{B,0},\quad
    p_2=v_{A,0}v_{B,2}+v_{A,1}v_{B,1}+v_{A,2}v_{B,0},
    \]
    \[
    p_3=v_{A,1}v_{B,2}+v_{A,2}v_{B,1},\qquad
    p_4=v_{A,2}v_{B,2},
    \]
    twelve products over the nine pairs $v_{A,i}v_{B,j}$:
    \begin{align*}
      &\rdf{\dsep{MEM},\fp\cdot o_X,r_X,v_{X,0},v_{X,1},v_{X,2}}\quad(X\in\{A,B\}),\\
      &\rdf{\dsep{MEM},\fp\cdot o_C,r_C,p_0+p_3,p_1+p_3+p_4,p_2+p_4}.
    \end{align*}
  \end{itemize}
\end{itemize}

\subsection{\texttt{SET\_CONSTANT}}\label{sec:tab-set}

\texttt{SET\_CONSTANT}\ $[o,k_0,k_1,k_2]$ asserts $\loc{o}=k_0+k_1y+k_2y^{2}$.
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operand $o$; immediate $(k_i)_{i\in\{0,1,2\}}$; memory count $r$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} $\rdf{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{SET},o,k_0,k_1,k_2,0}$.
  \item \textbf{Memory:} $\rdf{\dsep{MEM},\fp\cdot o,r,k_0,k_1,k_2}$.
  \end{itemize}
\end{itemize}

\subsection{\texttt{DEREF}}\label{sec:tab-deref}

\texttt{DEREF}\ $[o_1,o_2,o_3;\,\dmode]$ reads a pointer and asserts the cell it points to. The row reads three cells:
\begin{itemize}
\item the \emph{pointer} $p$, at $\fp\cdot o_1$;
\item the \emph{target} $v_2$, at the dereferenced address $p\cdot o_2$;
\item the \emph{local} value $v_3$, at $\fp\cdot o_3$.
\end{itemize}
The store asserts that $v_2$ equals the source chosen by the mode $\dmode$ (\S\ref{sec:vm}): the local cell $v_3$, the return address $\gen^{2}\cdot\pc$, or the frame pointer $\fp$. The mode is two boolean flags $(f_{\pc},f_{\fp})$: $(0,0)$ for $\texttt{deref\_cell}[o_3]$, $(1,0)$ for $\texttt{deref\_pc}$, $(0,1)$ for $\texttt{deref\_fp}$.
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_1,o_2,o_3$; flags $f_{\pc},f_{\fp}$; pointer $p$ (a single $\K$-limb); the local value $(v_{3,i})_{i\in\{0,1,2\}}$; memory counts $r_1,r_2,r_3$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} $\rdf{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{DRF},o_1,o_2,o_3,f_{\pc},f_{\fp}}$.
  \item \textbf{Memory}, the target giving $v_3$, $\gen^{2}\cdot\pc$ or $\fp$ at the three flag settings:
    \begin{align*}
      &\rdf{\dsep{MEM},\fp\cdot o_1,r_1,p,0,0},\\
      &\rdf{\dsep{MEM},\fp\cdot o_3,r_3,v_{3,0},v_{3,1},v_{3,2}},\\
      &\rdf{\dsep{MEM},p\cdot o_2,r_2,\bar f\,v_{3,0}+f_{\pc}\,(\gen^{2}\cdot\pc)+f_{\fp}\,\fp,\ \bar f\,v_{3,1},\ \bar f\,v_{3,2}},
      \qquad \bar f=1+f_{\pc}+f_{\fp}.
    \end{align*}
  \end{itemize}
\end{itemize}

\subsection{\texttt{JUMP}}\label{sec:tab-jump}

\texttt{JUMP}\ $[o_c,o_d,o_f]$ performs a conditional jump on whether the condition $v_{\mathrm{cond}}=\loc{o_c}$ is nonzero. It transfers to $\pc\gets v_{\pc} = \loc{o_d}$, $\fp\gets v_{\fp} = \loc{o_f}$ when $v_{\mathrm{cond}}\neq0$, and defaults to $\pc\gets\gen\cdot\pc$, $\fp\gets\fp$ otherwise. $v_{\mathrm{cond}}, v_{\fp}, v_{\pc}$ must be $\K$-valued. The branch is taken according to a boolean indicator column $b=[\,v_{\mathrm{cond}}\neq0\,]$, certified by a prover-supplied inverse $w$ and 2 constraints:

\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_c,o_d,o_f$; the condition $v_{\mathrm{cond}}$, the destination $v_{\pc}$ and the frame $v_{\fp}$; inverse $w$ (honest prover sets $w = v_{\mathrm{cond}}^{-1}$ if $v_{\mathrm{cond}}\neq0$, otherwise $w=0$); boolean indicator $b$; memory counts $r_c,r_d,r_f$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:}  $b=v_{\mathrm{cond}}\cdot w$ and $v_{\mathrm{cond}}\,(b+1)=0$
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},b\,v_{\pc}+b\,(\gen\cdot\pc)+\gen\cdot\pc,\;b\,v_{\fp}+b\,\fp+\fp}$.
  \item \textbf{Bytecode:} $\rdf{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{JMP},o_c,o_d,o_f,0,0}$.
  \item \textbf{Memory:}
    \begin{align*}
      &\rdf{\dsep{MEM},\fp\cdot o_c,r_c,v_{\mathrm{cond}},0,0},\\
      &\rdf{\dsep{MEM},\fp\cdot o_d,r_d,v_{\pc},0,0},\\
      &\rdf{\dsep{MEM},\fp\cdot o_f,r_f,v_{\fp},0,0}.
    \end{align*}
  \end{itemize}
\end{itemize}

\subsection{\texttt{KECCAK}}\label{sec:tab-keccak}

One row is one permutation (\S\ref{sec:vm}), proved by the GKR of Annex~\ref{annex:keccak}, of a \texttt{KECCAK} or a \texttt{KECCAK\_M}. A row has $120$ columns:
\begin{itemize}
\item $\pc$, $\fp$, the eight operand offsets, the two layout flags and $\mathit{free}$, and $27$ read counts, one per cell read or written and one for the bytecode.
\item $\mathit{mk}$, $1$ on a \texttt{KECCAK\_M} row; $b_8$, the low limb of the cell $\loc{o_{\mathit{b8}}}$ as read; the direction bit $s$; and $A_0$, $A_1$, the offsets the first two cells are read at.
\item $\kin{\ell}$, the $25$ lanes of the state the row reads from memory.
\item $\kmid{\ell}$, the $25$ lanes of the state after $12$ rounds. Committing it is an optimization and nothing more: it halves the depth of the reduction. No flush reads it.
\item $\kout{\ell}$, the $25$ lanes the row writes back. These are \emph{virtual}: the flushes read them, but nothing commits them, and the claims the table sumcheck leaves on them are deferred to the GKR, which reduces them to claims on $\kin{\ell}$ and $\kmid{\ell}$.
\end{itemize}

A limb holding lane $a$ in the even layout and $b$ in the odd one is the degree-$2$ form $a+\mathit{odd}\,(a+b)$, a limb holding no lane being the column $\mathit{free}$ when read and $0$ when written. The top limb is a literal $0$.
\begin{itemize}
\item \textbf{Constraints:} the permutation is proved by the GKR of Annex~\ref{annex:keccak}; eight identities of degree $2$ select the Merkle step:
  \begin{align*}
    &\mathit{mk}(\mathit{mk}+1)=0, \qquad \mathit{mk}\cdot\mathit{odd}_{\mathit{in}}=0, \qquad b_8+u_8+\mathit{mk}\,(b_8+\kin{16})=0, \qquad \mathit{mk}\,(\kin{16}+\texttt{0x80}\cdot2^{56})=0,\\
    &s+\mathit{mk}\cdot b_8=0, \qquad s(s+1)=0, \qquad A_0+o_{m_0}+s\,(o_{m_0}+o_{m_2})=0, \qquad A_1+o_{m_1}+s\,(o_{m_1}+o_{m_3})=0,
  \end{align*}
  $u_8=\kin{16}+\mathit{odd}_{\mathit{in}}(\kin{16}+\kin{15})$ being the form below. So $b_8$ is that lane on a \texttt{KECCAK} row and $s=0$; on a \texttt{KECCAK\_M} row the layout is even, lane $16$ is the padding byte, and $s=b_8$ is a bit.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} $\rdf{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{KEC}+\mathit{mk}\,(\opc{KEC}+\gen^{7}),o_{m_0},o_{m_1},o_{m_2},o_{m_3},o_{\mathit{tail}},o_{\mathit{b8}},o_{\mathit{cap}},o_{\mathit{out}},\mathit{odd}_{\mathit{in}},\mathit{odd}_{\mathit{out}}}$.
  \item \textbf{Memory:} cell $c<13$ is read at $a_c$ under $\mathit{odd}_{\mathit{in}}$ and written at $\gen^{c}\fp\cdot o_{\mathit{out}}$ under $\mathit{odd}_{\mathit{out}}$, where
    \[
      a_c=\begin{cases}
        \fp\cdot A_c & c<2,\\
        \fp\cdot(A_{c-2}+o_{m_{c-2}}+o_{m_c}) & 2\le c<4,\\
        \gen^{c-4}\fp\cdot o_{\mathit{tail}} & 4\le c<8,\\
        \fp\cdot o_{\mathit{b8}} & c=8,\\
        \gen^{c-9}\fp\cdot o_{\mathit{cap}} & 9\le c<13,
      \end{cases}
    \]
    \begin{align*}
      &\rdf{\dsep{MEM},a_c,r_c,u_c,v_c,0} &&(c<9),\\
      &\rdf{\dsep{MEM},a_c,r_c,\kin{2c-1},\kin{2c},0} &&(9\le c<13),
    \end{align*}
    the writes the same with $\kout{\ell}$ and $r_{13+c}$. The rate cells take the forms $u_c,v_c$ above, but for the read cell $8$, whose low limb is $b_8$; the four capacity cells hold lanes $17$ to $24$ in either layout. With $s=0$ the four $m$ cells are read at $\fp\cdot o_{m_c}$, with $s=1$ the two pairs swap.
  \end{itemize}
\end{itemize}

\subsection{\texttt{DOT}}\label{sec:tab-dot}

\texttt{DOT}\ $[o_x,o_p,o_w,o_{\mathit{acc}},o_{\mathit{out}}]$ asserts $\loc{o_{\mathit{out}}}=\loc{o_{\mathit{acc}}}+\sum_{j<4}w_j\,x_j$, an affine form of four $\K$-lanes with $\E$ weights (\S\ref{sec:vm}). It is the part of a verifier that is linear in the data it opens: once a child proof's transcript is fixed, the weights a recursive verifier puts on an opened row are words it has computed, and the check is affine in the row (\S\ref{sec:recursion}).
\begin{itemize}
\item \textbf{Columns:} $\pc,\fp$; operands $o_x,o_p,o_w,o_{\mathit{acc}},o_{\mathit{out}}$; the four lanes $x_0,\dots,x_3$; the pointer $p$ (a single $\K$-limb); the weights $(w_{j,i})_{j<4,\,i\in\{0,1,2\}}$; the accumulator $(a_i)_{i\in\{0,1,2\}}$; memory counts $r_{x,0},r_{x,1}$, $r_p$, $r_{w,0},\dots,r_{w,3}$, $r_{\mathit{acc}}$, $r_{\mathit{out}}$; bytecode count $r_{\mathrm{bc}}$.
\item \textbf{Constraints:} none.
\item \textbf{Flushes:}
  \begin{itemize}
  \item \textbf{State:} \pull\ $\tup{\dsep{ST},\pc,\fp}$, \push\ $\tup{\dsep{ST},\gen\cdot\pc,\fp}$.
  \item \textbf{Bytecode:} $\rdf{\dsep{BC},\pc,r_{\mathrm{bc}},\opc{DOT},o_x,o_p,o_w,o_{\mathit{acc}},o_{\mathit{out}}}$.
  \item \textbf{Memory}, the result carrying, limb by limb, the degree-two form $a_i+\sum_{j<4}w_{j,i}\,x_j$ (a $\K$-lane times an $\E$ word needs no tower reduction):
    \begin{align*}
      &\rdf{\dsep{MEM},\gen^{k}\fp\cdot o_x,r_{x,k},x_{2k},x_{2k+1},0}\quad(k<2),\\
      &\rdf{\dsep{MEM},\fp\cdot o_p,r_p,p,0,0},\\
      &\rdf{\dsep{MEM},\gen^{j}p\cdot o_w,r_{w,j},w_{j,0},w_{j,1},w_{j,2}}\quad(j<4),\\
      &\rdf{\dsep{MEM},\fp\cdot o_{\mathit{acc}},r_{\mathit{acc}},a_0,a_1,a_2},\\
      &\rdf{\dsep{MEM},\fp\cdot o_{\mathit{out}},r_{\mathit{out}},\textstyle a_0+\sum_j w_{j,0}x_j,\ a_1+\sum_j w_{j,1}x_j,\ a_2+\sum_j w_{j,2}x_j}.
    \end{align*}
  \end{itemize}
\end{itemize}
The lane cells' top limbs are literal zeros and the lanes are $\K$ columns, so the bus alone proves each cell canonical and both its lanes in $\K$: a run that \texttt{DOT} weighs and \texttt{KECCAK} hashes needs no packing instructions (\S\ref{sec:prog-subfield-checks}). The weights are read through a pointer, as \texttt{DEREF} reads its target, and the bus forces $p\cdot o_w\cdot\gen^{j}$ to be an address (Corollary~\ref{cor:addrrange}).

The table's least log height is $13$, as \texttt{KECCAK}'s is that of a block of steps. A recursive verifier opens each column of a child proof through its stacking selector, over the child's position bits above the column's height, and decomposes each bus block's offset the same way, so an empty table would cost it the most; the floor caps that cost at the price of $2^{13}$ padding rows. The width, four lanes, balances the same fixed cost per column and per block against the rows a row saves.
